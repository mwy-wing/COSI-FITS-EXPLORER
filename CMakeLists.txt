cmake_minimum_required(VERSION 3.20)
# Public versioning began at 0.1.0; prototype labels were never published.
if(CMAKE_HOST_APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0" CACHE STRING "Minimum supported macOS version")
endif()
project(COSIExplorer VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CFITSIO REQUIRED IMPORTED_TARGET cfitsio)
add_library(cosi_data src/Data.cpp src/Workflow.cpp)
target_include_directories(cosi_data PUBLIC src)
target_link_libraries(cosi_data PUBLIC PkgConfig::CFITSIO)
target_compile_options(cosi_data PRIVATE -Wall -Wextra)
add_executable(cosi-inspect src/Inspect.cpp)
target_link_libraries(cosi-inspect PRIVATE cosi_data)
add_executable(cosi-tests tests/core_tests.cpp)
target_link_libraries(cosi-tests PRIVATE cosi_data)
enable_testing()
add_test(NAME data_readers COMMAND cosi-tests)
add_executable(cosi-workflow-tests tests/workflow_tests.cpp)
target_link_libraries(cosi-workflow-tests PRIVATE cosi_data)
add_test(NAME event_tracing_and_overlays COMMAND cosi-workflow-tests)
add_executable(cosi-make-demo EXCLUDE_FROM_ALL examples/generate-demo.cpp)
target_link_libraries(cosi-make-demo PRIVATE PkgConfig::CFITSIO)
add_executable(cosi-demo-tests tests/demo_tests.cpp)
target_link_libraries(cosi-demo-tests PRIVATE cosi_data)
add_test(NAME synthetic_demo COMMAND cosi-demo-tests "${CMAKE_CURRENT_SOURCE_DIR}/examples/synthetic-events.fits")
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  set_source_files_properties(resources/COSIExplorer.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  add_executable(COSIExplorer MACOSX_BUNDLE src/App.mm resources/COSIExplorer.icns)
  target_compile_options(COSIExplorer PRIVATE -fobjc-arc -Wall -Wextra -Wno-unused-parameter)
  target_link_libraries(COSIExplorer PRIVATE cosi_data "-framework Cocoa" "-framework UniformTypeIdentifiers")
  set_target_properties(COSIExplorer PROPERTIES
    MACOSX_BUNDLE_BUNDLE_NAME "COSI Explorer"
    MACOSX_BUNDLE_GUI_IDENTIFIER "org.local.cosiexplorer"
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist")
endif()

if(UNIX AND NOT APPLE)
  option(COSI_BUILD_GUI "Build the Linux Qt graphical app" ON)
  if(COSI_BUILD_GUI)
    find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Concurrent Test)
    add_library(cosi_qt src/qt/Explorer.cpp)
    target_include_directories(cosi_qt PUBLIC src/qt)
    target_compile_definitions(cosi_qt PUBLIC COSI_VERSION="${PROJECT_VERSION}")
    target_compile_options(cosi_qt PRIVATE -Wall -Wextra)
    target_link_libraries(cosi_qt PUBLIC cosi_data Qt6::Widgets Qt6::Concurrent)
    qt_add_resources(cosi_qt "cosi_artwork" PREFIX "/cosi" BASE resources
      FILES resources/COSIExplorer.png)
    add_executable(cosi-explorer src/qt/Main.cpp)
    target_link_libraries(cosi-explorer PRIVATE cosi_qt)
    add_executable(cosi-qt-tests tests/qt_tests.cpp)
    target_link_libraries(cosi-qt-tests PRIVATE cosi_qt Qt6::Test)
    add_test(NAME linux_gui COMMAND cosi-qt-tests "${CMAKE_CURRENT_SOURCE_DIR}/examples/synthetic-events.fits")
    set_tests_properties(linux_gui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120)
    include(GNUInstallDirs)
    install(TARGETS cosi-explorer cosi-inspect RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(FILES resources/linux/cosi-explorer.desktop DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/applications)
    install(FILES resources/COSIExplorer.png DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/icons/hicolor/1024x1024/apps RENAME cosi-explorer.png)
    install(FILES LICENSE THIRD_PARTY_NOTICES.txt DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/doc/cosi-explorer)
    set(CPACK_GENERATOR DEB)
    set(CPACK_PACKAGE_NAME cosi-explorer)
    set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
    set(CPACK_PACKAGE_CONTACT "WingYeung Ma <97164522+mwy-wing@users.noreply.github.com>")
    set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Graphical FITS and COSI event data viewer")
    set(CPACK_DEBIAN_PACKAGE_SECTION science)
    set(CPACK_DEBIAN_PACKAGE_HOMEPAGE "https://github.com/mwy-wing/COSI-FITS-EXPLORER")
    set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
    # Platform plugins are loaded at runtime and are invisible to shlibdeps.
    set(CPACK_DEBIAN_PACKAGE_DEPENDS "qt6-qpa-plugins")
    set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "qt6-wayland")
    set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
    set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
    include(CPack)
  endif()
endif()
